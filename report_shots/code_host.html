def detect_real_device():
    try:
        import torch
        if torch.cuda.is_available():
            return (torch.cuda.get_device_name(0), "torch")
    except Exception:
        pass
    try:
        import cupy
        if cupy.cuda.runtime.getDeviceCount() > 0:
            return ("cupy-cuda", "cupy")
    except Exception:
        pass
